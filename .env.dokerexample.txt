# --- Django core -------------------------------------------------------------
# Generate a long random value for production.
DJANGO_SECRET_KEY=your-SECRET_KEY
# Set to False for production.

DJANGO_ALLOWED_HOSTS=127.0.0.1,localhost
DJANGO_CSRF_TRUSTED_ORIGINS=http://localhost:8001,http://127.0.0.1:8001

#   python -c "from cryptography.fernet import Fernet; print(Fernet.generate_key().decode())"
DJANGO_FIELD_ENCRYPTION_KEY=your-fernet-encryption-key

# --- Database (as seen by the app) --------------------------------------------
# The host is the database service name inside the compose network.

DJANGO_DB_NAME=chaviclient
DJANGO_DB_USER=postgres
DJANGO_DB_PASSWORD=your-password

# --- Postgres container ---------------------------------------------------------
